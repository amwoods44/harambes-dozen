    ```javascript
    function renderKeeperTable(){
      var wrap=document.getElementById('con-table-wrap');
      if(!wrap)return;
      // Build player array from filtered teams
      var filtC=selConTeam?D.teams.filter(function(t){return t.team_name===selConTeam}):D.teams;
      var allP=[];
      filtC.forEach(function(t){
        t.players.forEach(function(p){
          var c=D.contracts&&(D.contracts[p.name]||D.contracts['_id_'+p.id]);
          if(!c)return;
          allP.push({
            name:p.name,
            id:p.id,
            team:t.team_name,
            rid:t.roster_id,
            pos:p.pos||'',
            yrs:c.yrs,
            contracted:c.contracted,
            tag:c.tag,
            exm:c.exm,
            ktc:c.ktc||0,
            rnk:c.rnk||9999,
            age:p.age||0
          });
        });
      });
      // Sort
      var col=conSortCol,dir=conSortDir;
      allP.sort(function(a,b){
        var av=a[col],bv=b[col];
        if(typeof av==='string'&&typeof bv==='string')return dir*av.localeCompare(bv);
        if(av===null||av===undefined)av=col==='ktc'?0:col==='yrs'?-1:'';
        if(bv===null||bv===undefined)bv=col==='ktc'?0:col==='yrs'?-1:'';
        return dir*(av>bv?1:av<bv?-1:0);
      });
      // Sort indicator
      function si(c){return c===col?(dir===-1?' \u25BC':' \u25B2'):'';};
      // Build table
      var h='<table class="keeper-tbl">';
      h+='<thead><tr>';
      h+='<th onclick="sortKeeperSheet(\'name\')">PLAYER'+si('name')+'</th>';
      h+='<th onclick="sortKeeperSheet(\'team\')">TEAM'+si('team')+'</th>';
      h+='<th onclick="sortKeeperSheet(\'pos\')">POS'+si('pos')+'</th>';
      h+='<th onclick="sortKeeperSheet(\'yrs\')">YRS'+si('yrs')+'</th>';
      h+='<th onclick="sortKeeperSheet(\'tag\')">TAG'+si('tag')+'</th>';
      h+='<th onclick="sortKeeperSheet(\'exm\')">EXEMPT'+si('exm')+'</th>';
      h+='<th onclick="sortKeeperSheet(\'ktc\')">KTC'+si('ktc')+'</th>';
      h+='</tr></thead>';
      h+='<tbody>';
      allP.forEach(function(p,i){
        var yrCol=p.yrs>=4?'var(--a)':p.yrs>=2?'var(--y)':p.yrs===1?'var(--red-t)':'var(--t4)';
        var yrText=p.contracted?(p.yrs<=0?'EXP':p.yrs+'yr'):'--';
        var tagText=p.tag?esc(typeof p.tag==='string'?p.tag:'TAG'):'';
        var exmText=p.exm?esc(p.exm):'';
        h+='<tr class="'+(i%2===0?'keeper-even':'')+'">';
        h+='<td class="keeper-name"><span class="pp-trigger" data-pid="'+p.id+'">'+esc(p.name)+'</span>'+cpill(p.name)+(p.yrs===1?cbadge('RELEASABLE','red'):'')+'</td>';
        h+='<td>'+av(p.rid,16)+' <span class="keeper-team">'+esc(p.team).substring(0,14)+'</span></td>';
        h+='<td><span class="pl-pos pos-'+p.pos+'">'+p.pos+'</span></td>';
        h+='<td style="color:'+yrCol+';font-weight:700">'+yrText+'</td>';
        h+='<td>'+(tagText?cbadge(tagText,'y'):'')+'</td>';
        h+='<td>'+(exmText?cbadge(exmText,'pur'):'')+'</td>';
        h+='<td style="font-family:var(--fm);font-weight:700;color:var(--a)">'+(p.ktc?p.ktc.toLocaleString():'--')+'</td>';
        h+='</tr>';
      });
      h+='</tbody></table>';
      h+='<div style="font-family:var(--fb);font-size:11px;color:var(--t4);margin-top:8px;text-align:right">'+allP.length+' players \u2022 Sorted by '+conSortCol.toUpperCase()+(conSortDir===-1?' (desc)':' (asc)')+'</div>';
      wrap.innerHTML=h;
    }
    ```
