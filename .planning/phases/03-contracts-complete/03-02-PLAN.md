<!-- cpill function (line 2520) - DO NOT modify the signature -->
```javascript
function cpill(nm) {
  if (!D.contracts) return '';
  const c = D.contracts[nm];
  if (!c || !c.contracted) return '';
  const y = c.yrs;
  const cls = y >= 4 ? 'cpill-4' : y >= 2 ? 'cpill-23' : y === 1 ? 'cpill-1' : 'cpill-0';
  const lbl = y <= 0 ? 'EXP' : y + 'yr';
  const tag = c.tag ? 'T' : c.exm ? 'X' : '';
  return '<span class="cpill ' + cls + '">' + tag + lbl + '</span>';
}
```

```javascript
function cbadge(text, color) {
  var m = {'y':'245,158,11','blu':'59,130,246','pur':'139,92,246','red':'239,68,68','gray':'102,102,120'};
  var rgb = m[color] || '102,102,120';
  var cc = color==='y'?'var(--y)':color==='blu'?'var(--blu)':color==='pur'?'var(--pur)':color==='red'?'var(--red-t)':'var(--t4)';
  return '<span style="font-family:var(--fd);font-size:9px;font-weight:700;letter-spacing:1px;padding:2px 6px;border-radius:3px;background:rgba('+rgb+',.12);color:'+cc+'">'+text+'</span>';
}
```
