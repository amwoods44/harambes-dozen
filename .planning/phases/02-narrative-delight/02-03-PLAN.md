function badgeSvg(iconName,teamColor){
  var tc=teamColor||'#cc0000';
  var gid='bg'+Math.random().toString(36).substr(2,5);
  return '<svg width="48" height="48" viewBox="0 0 48 48" xmlns="http://www.w3.org/2000/svg">'
    +'<defs><radialGradient id="'+gid+'" cx="35%" cy="30%">'
    +'<stop offset="0%" stop-color="#f5e070"/>'
    +'<stop offset="40%" stop-color="#d4a843"/>'
    +'<stop offset="100%" stop-color="#8a6420"/>'
    +'</radialGradient></defs>'
    +'<circle cx="24" cy="24" r="22" fill="url(#'+gid+')" stroke="'+tc+'" stroke-width="2" stroke-opacity=".4"/>'
    +'<circle cx="24" cy="24" r="19" fill="none" stroke="#e8c65a" stroke-width=".5" stroke-opacity=".3"/>'
    +(icon?icon(iconName,20,'#2a1a00'):'')
    +'</svg>';
}

**computeLifetimeBadges(rosterIdOrName)** — returns array of `{id, label, desc, icon, tc}` for a team's lifetime achievements:

```javascript
function computeLifetimeBadges(rosterIdOrName){
  var rid=typeof rosterIdOrName==='number'?rosterIdOrName:null;
  var teamName=typeof rosterIdOrName==='string'?rosterIdOrName:(D.rid_to_name[rid]||'');
  if(!teamName)return[];
  if(!rid){var t=D.teams.find(function(x){return x.team_name===teamName;});rid=t?t.roster_id:null;}
  var badges=[];
  var tc=(TC&&TC[rid])?TC[rid].p:'#cc0000';

  // Champion — won at least one championship
  if(D.champions.some(function(c){return c.champion===teamName;})){
    var titles=D.champions.filter(function(c){return c.champion===teamName;}).length;
    badges.push({id:'champion',label:'Dynasty Champ',desc:titles+'x Champion',icon:'trophy',tc:tc});
  }

  // Century Club — scored 200+ in any week across all seasons
  var allScores=[];
  SEASON_YEARS.forEach(function(yr){
    var key=yr===CURRENT_YEAR?'weekly_scores':'weekly_scores_'+yr;
    var sc=D[key];
    if(sc&&rid&&sc[rid])allScores=allScores.concat(sc[rid]);
  });
  if(allScores.some(function(p){return p>=200;})){
    badges.push({id:'century',label:'Century Club',desc:'Scored 200+ in a week',icon:'fire',tc:tc});
  }

  // Wheeler-Dealer — 5+ lifetime trades
  var act=D.activity[teamName]||{};
  if((act.trades||0)>=5){
    badges.push({id:'dealer',label:'Wheeler-Dealer',desc:(act.trades||0)+' lifetime trades',icon:'trade',tc:tc});
  }

  // Ghost — fewest total moves among all managers
  var allTotals=Object.values(D.activity).map(function(a){return(a.trades||0)+(a.moves||0);}).sort(function(a,b){return a-b;});
  var myTotal=(act.trades||0)+(act.moves||0);
  if(myTotal<=allTotals[1]){
    badges.push({id:'ghost',label:'The Ghost',desc:'Fewest lifetime moves',icon:'ghost',tc:tc});
  }

  // Point Machine — franchise all-time points leader or top-2
  var fr=(D.franchise||[]).find(function(f){return f.name===teamName;});
  if(fr){
    var topPts=Math.max.apply(null,(D.franchise||[]).map(function(f){return f.total_pts||0;}));
    if(fr.total_pts>=topPts*0.97){
      badges.push({id:'machine',label:'Point Machine',desc:'All-time points leader',icon:'bolt',tc:tc});
    }
  }

  // Dynasty Builder — youngest current roster (avg age <= 24.5)
  var team=D.teams.find(function(x){return x.team_name===teamName;});
  if(team){
    var ages=team.players.filter(function(p){return p.age&&p.pos!=='DEF';}).map(function(p){return p.age;});
    var avgAge=ages.length?ages.reduce(function(a,b){return a+b;},0)/ages.length:28;
    if(avgAge<=24.5){badges.push({id:'builder',label:'Dynasty Builder',desc:'Avg age '+avgAge.toFixed(1),icon:'seedling',tc:tc});}
  }

  // Draft Guru — 3+ draft picks became starters
  if(D.draft_hits){
    var hits=Object.values(D.draft_hits).flat().filter(function(h){return h.status==='starter'&&h.drafter===teamName;});
    if(hits.length>=3){badges.push({id:'guru',label:'Draft Guru',desc:hits.length+' picks became starters',icon:'draft',tc:tc});}
  }

  // Iron Man — most total transactions (top player in moves)
  var allActArr=Object.entries(D.activity).map(function(e){return{name:e[0],total:(e[1].trades||0)+(e[1].moves||0)+(e[1].draft_picks||0)};}).sort(function(a,b){return b.total-a.total;});
  if(allActArr.length&&allActArr[0].name===teamName){
    badges.push({id:'ironman',label:'Iron Man',desc:'Most active manager',icon:'target',tc:tc});
  }

  return badges;
}
```
