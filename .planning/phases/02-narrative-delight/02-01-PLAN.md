function loadCountUp(cb){

  s.src='https://cdn.jsdelivr.net/npm/countup.js@2.10.0/dist/countUp.umd.js';

}

**Step B — Replace animateCounters() entirely.** The existing function from `function animateCounters(scope){` through its closing `}` (currently lines 5012-5030) must be replaced with:

```javascript
function animateCounters(scope){
  if(!window.countUp){loadCountUp(function(){animateCounters(scope);});return;}
  (scope||document).querySelectorAll('.count-up').forEach(function(el){
    var target=parseFloat(el.getAttribute('data-target'));
    if(isNaN(target))return;
    var decimals=parseInt(el.getAttribute('data-decimals')||'0');
    var suffix=el.getAttribute('data-suffix')||'';
    var c=new window.countUp.CountUp(el,target,{
      duration:1.5,
      decimalPlaces:decimals,
      suffix:suffix,
      useEasing:true,
      easingFn:function(t,b,c,d){return c*(1-Math.pow(1-t/d,3))+b;}
    });
    if(!c.error)c.start();
  });
}
```
