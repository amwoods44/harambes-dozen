**Step A — Add getCurrentNFLWeek() helper.** Insert this function immediately before buildHistoryFacts():

```javascript
function getCurrentNFLWeek(){
  var now=new Date();
  // Find first Thursday of September for a given year
  function septKickoff(yr){
    var d=new Date(yr,8,1); // Sep 1
    var dow=d.getDay();
    var daysToThur=(4-dow+7)%7;
    return new Date(yr,8,1+daysToThur);
  }
  var kickoff=septKickoff(now.getFullYear());
  var diff=now-kickoff;
  if(diff<0){kickoff=septKickoff(now.getFullYear()-1);diff=now-kickoff;}
  return Math.max(1,Math.min(18,Math.floor(diff/(7*24*60*60*1000))+1));
}
```

**Step B — Replace buildHistoryFacts() entirely.** The replacement function:
1. Gets the current NFL week via getCurrentNFLWeek()
2. Searches all historical seasons (excluding CURRENT_YEAR) for games played in that same week
3. Picks the single most significant fact by highest score in that week
4. Falls back to the existing rotating ticker behavior if no match found
5. Renders an upgraded card with vintage ESPN treatment

```javascript
function buildHistoryFacts(){
  var el=document.getElementById('hero-hist');
  if(!el)return;

  var week=getCurrentNFLWeek();
  var bestFact=null;
  var bestScore=0;

  // Search historical seasons for the same week number
  var histYears=SEASON_YEARS.filter(function(yr){return yr!==CURRENT_YEAR;});
  histYears.forEach(function(yr){
    var mwKey='matchup_weeks_'+yr;
    var mw=D[mwKey]||{};
    var games=mw[week]||[];
    games.forEach(function(g){
      var highScore=Math.max(g.team_a.points,g.team_b.points);
      var highTeam=g.team_a.points>=g.team_b.points?g.team_a.name:g.team_b.name;
      if(highScore>bestScore){
        bestScore=highScore;
        bestFact={yr:yr,wk:week,team:highTeam,pts:highScore.toFixed(2),type:'high'};
      }
    });
  });

  if(bestFact){
    el.innerHTML=
      '<div class="hero-hist-vintage">'+
        '<div class="hero-hist-eyebrow">THIS WEEK IN LEAGUE HISTORY \u2022 WEEK '+bestFact.wk+'</div>'+
        '<div class="hero-hist-headline">'+
          '<span class="hero-hist-team">'+esc(bestFact.team)+'</span>'+
          '<span class="hero-hist-verb"> dropped </span>'+
          '<span class="hero-hist-pts">'+bestFact.pts+'</span>'+
          '<span class="hero-hist-verb"> pts in '+bestFact.yr+'</span>'+
        '</div>'+
      '</div>';
    return;
  }

  // Fallback: rotating ticker (existing behavior preserved)
  var facts=[
    D.champions[0]?'\u2605 '+D.champions[0].champion+' \u2014 '+D.champions[0].year+' Champion':'',
    D.records.highest_score?'All-time high: '+D.records.highest_score.pts+' pts by '+tn(D.records.highest_score.roster_id):'',
    D.records.closest_game?'Closest ever: '+D.records.closest_game.diff+' pts \u2022 Wk '+D.records.closest_game.week:'',
    D.records.biggest_blowout?'Biggest blowout: '+D.records.biggest_blowout.diff+' pts \u2022 Wk '+D.records.biggest_blowout.week:'',
    D.trades.length+' total trades \u2022 All Seasons',
    D.champions[1]?'\u2605 '+D.champions[1].champion+' \u2014 '+D.champions[1].year+' Champion over '+D.champions[1].runner_up:'',
  ].filter(Boolean);
  if(!facts.length)return;
  var idx=0;
  function show(){
    var inner=el.querySelector('.hero-hist-txt');
    if(!inner)return;
    inner.classList.add('fade');
    setTimeout(function(){inner.textContent=facts[idx%facts.length];inner.classList.remove('fade');idx++;},350);
  }
  el.innerHTML='<div class="hero-hist-inner"><span class="hero-hist-lbl">LEAGUE LORE</span><span class="hero-hist-txt">'+facts[0]+'</span></div>';
  idx=1;
  setInterval(show,4000);
}
```
