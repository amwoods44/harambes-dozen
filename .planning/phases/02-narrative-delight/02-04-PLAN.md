**Step A — Add buildPowerMoves() function.** Insert before the `// ═══ COUNTUP LOADER ═══` comment (or wherever loadCountUp is after Plan 01):

```javascript
function buildPowerMoves(year){
  var mwKey=year===CURRENT_YEAR?'matchup_weeks':'matchup_weeks_'+year;
  var mw=D[mwKey]||{};
  var weeks=Object.keys(mw).map(Number).sort(function(a,b){return a-b;});
  var stories=[];

  weeks.forEach(function(wk){
    var games=mw[wk]||[];
    if(!games.length)return;

    // Total points this week
    var totalPts=games.reduce(function(s,g){return s+g.team_a.points+g.team_b.points;},0);

    // Highest single score this week
    var topGame=games.slice().sort(function(a,b){
      return Math.max(b.team_a.points,b.team_b.points)-Math.max(a.team_a.points,a.team_b.points);
    })[0];
    if(topGame){
      var high=Math.max(topGame.team_a.points,topGame.team_b.points);
      var highTeam=topGame.team_a.points>=topGame.team_b.points?topGame.team_a.name:topGame.team_b.name;
      if(high>=180){
        stories.push({wk:wk,type:'explosion',pts:high,
          headline:esc(highTeam)+' dropped <strong>'+high+'</strong> in Week '+wk+'. Nobody saw it coming.'});
      }
    }

    // Closest game (heartbreak)
    var closest=games.slice().sort(function(a,b){return a.diff-b.diff;})[0];
    if(closest&&closest.diff<=2){
      stories.push({wk:wk,type:'heartbreak',pts:closest.diff,
        headline:'Week '+wk+': <strong>'+esc(closest.loser.name)+'</strong> lost by <strong>'+closest.diff.toFixed(2)+'</strong>. That\'s a bad beat.'});
    }

    // Biggest blowout (only if > 40 pts margin)
    var biggestBlowout=games.slice().sort(function(a,b){return b.diff-a.diff;})[0];
    if(biggestBlowout&&biggestBlowout.diff>=40){
      stories.push({wk:wk,type:'blowout',pts:biggestBlowout.diff,
        headline:'<strong>'+esc(biggestBlowout.winner.name)+'</strong> steamrolled <strong>'+esc(biggestBlowout.loser.name)+'</strong> by '+biggestBlowout.diff.toFixed(1)+' in Week '+wk+'. Not close.'});
    }

    // High-scoring week (total > 1800 for 12-team league)
    if(totalPts>=1800){
      stories.push({wk:wk,type:'highweek',pts:totalPts,
        headline:'The week the league woke up. <strong>'+Math.round(totalPts)+'</strong> total points scored in Week '+wk+'.'});
    }
  });

  // Trades from D.unified_trades for this season
  var yearNum=parseInt(year);
  (D.unified_trades||[]).filter(function(t){return parseInt(t.yr)===yearNum||parseInt(t.s)===yearNum;}).forEach(function(tr){
    var sA=tr.sides&&tr.sides[0],sB=tr.sides&&tr.sides[1];
    if(sA&&sB&&(sA.players.length+sB.players.length)>=3){
      stories.push({wk:tr.wk||0,type:'trade',pts:0,
        headline:'Week '+(tr.wk||'?')+': <strong>'+esc(sA.name)+'</strong> and <strong>'+esc(sB.name)+'</strong> struck a deal. The jury is still out.'});
    }
  });

  // Sort by week, then by pts descending (most dramatic first within week)
  stories.sort(function(a,b){return a.wk-b.wk||b.pts-a.pts;});

  // Cap: max 2 stories per week, top 12 total
  var perWeek={};
  return stories.filter(function(s){
    perWeek[s.wk]=(perWeek[s.wk]||0)+1;
    return perWeek[s.wk]<=2;
  }).slice(0,12);
}
```
